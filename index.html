import openpyxl import json from pathlib import Path excel_file = "vartani_questions_corrected.xlsx" wb = openpyxl.load_workbook(excel_file, data_only=True) ws = wb.active questions = [] for row in ws.iter_rows(min_row=2, values_only=True): if not row[0]: continue question = str(row[0]).strip() options = [ str(x or "").strip() for x in row[1:5] ] answer = str( row[5] or "" ).strip().upper() if answer in ["A", "B", "C", "D"]: questions.append({ "q": question, "o": options, "a": ord(answer) - 65 }) data = json.dumps( questions, ensure_ascii=False ) html = """ Hindi Vartani Quiz

⏱️ 10
वर्तनी शुद्धि (पिछले सालों में RO/ARO की परीक्षा में पूछे गए महत्वपूर्ण प्रश्न)
""" html = html.replace( "__QUESTIONS__", data ) Path("index.html").write_text( html, encoding="utf-8" ) print() print("====================================") print("QUIZ READY!") print( "Total Questions:", len(questions) ) print( "index.html created successfully." ) print("====================================")
